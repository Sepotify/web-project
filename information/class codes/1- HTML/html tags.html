1. Page Structure Tags

These tags form the skeleton of every HTML document.

<!DOCTYPE html>        <!-- Declares HTML5 -->
<html lang="en">       <!-- Root element -->
  <head>               <!-- Metadata container -->
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Page Title</title>
    <link rel="stylesheet" href="style.css">
    <script src="app.js"></script>
  </head>
  <body>               <!-- Visible content goes here -->
  </body>
</html>

4. Links & Media

<!-- Audio -->

<!-- Iframe -->

7. Semantic & Layout Tags

<header>   <!-- Page or section header -->
<nav>      <!-- Navigation links -->
<main>     <!-- Main content area -->
<section>  <!-- Thematic section -->
<article>  <!-- Self-contained content -->
<aside>    <!-- Sidebar / related content -->
<footer>   <!-- Page or section footer -->
<figure>   <!-- Image with caption -->
  <img src="..." alt="...">
  <figcaption>Caption here</figcaption>
</figure>
<div>      <!-- Generic block container -->
<span>     <!-- Generic inline container -->

Semantic tags give meaning to your layout. Search engines and screen readers use them to understand page structure.
